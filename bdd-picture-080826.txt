| piyes_ledger_get_or_create_customer_account | CREATE OR REPLACE FUNCTION public.piyes_ledger_get_or_create_customer_account(p_customer_user_id text, p_name text DEFAULT 'Compte client'::text, p_piyes_account_id text DEFAULT NULL::text, p_piyes_user_id text DEFAULT NULL::text, p_initial_balance_cents bigint DEFAULT 0)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_id uuid;
BEGIN
  SELECT id INTO v_id FROM ledger_account WHERE customer_user_id = p_customer_user_id;
  IF v_id IS NULL THEN
    INSERT INTO ledger_account (code, name, type, customer_user_id, piyes_account_id, piyes_user_id)
    VALUES ('110' || right(md5(p_customer_user_id), 8), p_name, 'ASSET',
            p_customer_user_id, p_piyes_account_id, p_piyes_user_id)
    ON CONFLICT DO NOTHING
    RETURNING id INTO v_id;

    IF v_id IS NULL THEN
      SELECT id INTO v_id FROM ledger_account WHERE customer_user_id = p_customer_user_id;
    END IF;

    INSERT INTO ledger_account_balance (ledger_account_id, balance_cents)
    VALUES (v_id, p_initial_balance_cents)
    ON CONFLICT (ledger_account_id) DO UPDATE
      SET balance_cents = EXCLUDED.balance_cents;
  ELSE
    IF p_piyes_account_id IS NOT NULL OR p_piyes_user_id IS NOT NULL THEN
      UPDATE ledger_account
         SET piyes_account_id = COALESCE(p_piyes_account_id, piyes_account_id),
             piyes_user_id    = COALESCE(p_piyes_user_id, piyes_user_id)
       WHERE id = v_id;
    END IF;
  END IF;

  RETURN v_id;
END;
$function$

| piyes_ledger_post                           | CREATE OR REPLACE FUNCTION public.piyes_ledger_post(p_idempotency_key text, p_type text, p_customer_user_id text, p_amount_cents bigint, p_debit_account_id uuid, p_credit_account_id uuid, p_description text, p_external_ref text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_order payment_order%ROWTYPE;
  v_journal_id uuid;
  v_new_debit_balance bigint;
  v_new_credit_balance bigint;
BEGIN
  -- Idempotence : un même key = même ordre (replay possible)
  IF p_idempotency_key IS NOT NULL THEN
    SELECT * INTO v_order FROM payment_order WHERE idempotency_key = p_idempotency_key;
    IF FOUND THEN
      RETURN jsonb_build_object(
        'payment_order_id', v_order.id,
        'journal_entry_id', v_order.journal_entry_id,
        'status', v_order.status,
        'replay', true
      );
    END IF;
  END IF;

  -- Création de l'ordre (statut SUBMITTED), verrouillé par la clé unique
  BEGIN
    INSERT INTO payment_order (idempotency_key, customer_user_id, type, status, amount_cents,
                               debit_account_id, credit_account_id)
    VALUES (p_idempotency_key, p_customer_user_id, p_type, 'SUBMITTED', p_amount_cents,
            p_debit_account_id, p_credit_account_id)
    RETURNING * INTO v_order;
  EXCEPTION WHEN unique_violation THEN
    SELECT * INTO v_order FROM payment_order WHERE idempotency_key = p_idempotency_key;
    IF NOT FOUND THEN RAISE EXCEPTION 'CONCURRENCE_ORDER_CREATE'; END IF;
    RETURN jsonb_build_object(
      'payment_order_id', v_order.id,
      'journal_entry_id', v_order.journal_entry_id,
      'status', v_order.status,
      'replay', true
    );
  END;

  -- Écriture au journal (débit/crédit atomique)
  BEGIN
    v_journal_id := post_journal(
      p_external_ref,
      p_description,
      jsonb_build_array(
        jsonb_build_object('ledger_account_id', p_debit_account_id,  'direction', 'DEBIT',  'amount_cents', p_amount_cents),
        jsonb_build_object('ledger_account_id', p_credit_account_id, 'direction', 'CREDIT', 'amount_cents', p_amount_cents)
      )
    );
  EXCEPTION WHEN OTHERS THEN
    UPDATE payment_order SET status = 'FAILED', updated_at = now() WHERE id = v_order.id;
    RAISE;
  END;

  UPDATE payment_order
     SET status = 'SETTLED', journal_entry_id = v_journal_id, updated_at = now()
   WHERE id = v_order.id;

  -- Nouveaux soldes du ledger
  SELECT balance_cents INTO v_new_debit_balance
    FROM ledger_account_balance WHERE ledger_account_id = p_debit_account_id;
  SELECT balance_cents INTO v_new_credit_balance
    FROM ledger_account_balance WHERE ledger_account_id = p_credit_account_id;

  -- Mirroring piYès : Account + User (lecture seule côté frontend)
  UPDATE "Account" SET balance = v_new_debit_balance, "updatedAt" = now()
   WHERE id = (SELECT piyes_account_id FROM ledger_account WHERE id = p_debit_account_id);
  UPDATE "User" SET balance = v_new_debit_balance, "updatedAt" = now()
   WHERE id = (SELECT piyes_user_id FROM ledger_account WHERE id = p_debit_account_id);

  UPDATE "Account" SET balance = v_new_credit_balance, "updatedAt" = now()
   WHERE id = (SELECT piyes_account_id FROM ledger_account WHERE id = p_credit_account_id);
  UPDATE "User" SET balance = v_new_credit_balance, "updatedAt" = now()
   WHERE id = (SELECT piyes_user_id FROM ledger_account WHERE id = p_credit_account_id);

  RETURN jsonb_build_object(
    'payment_order_id', v_order.id,
    'journal_entry_id', v_journal_id,
    'status', 'SETTLED',
    'replay', false,
    'debit_balance', v_new_debit_balance,
    'credit_balance', v_new_credit_balance
  );
END;

| post_journal                                | CREATE OR REPLACE FUNCTION public.post_journal(p_external_ref text, p_description text, p_lines jsonb)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_entry_id uuid := gen_random_uuid();
  v_debit bigint := 0;
  v_credit bigint := 0;
  l record;
BEGIN
  -- Validation : montants > 0, directions valides, débits = crédits
  FOR l IN SELECT (line->>'ledger_account_id')::uuid AS ledger_account_id,
                  (line->>'direction')::text         AS direction,
                  (line->>'amount_cents')::bigint     AS amount_cents
           FROM jsonb_array_elements(p_lines) AS line
  LOOP
    IF l.amount_cents IS NULL OR l.amount_cents <= 0 THEN
      RAISE EXCEPTION 'MONTANT_INVALIDE';
    END IF;
    IF l.direction = 'DEBIT' THEN
      v_debit := v_debit + l.amount_cents;
    ELSIF l.direction = 'CREDIT' THEN
      v_credit := v_credit + l.amount_cents;
    ELSE
      RAISE EXCEPTION 'DIRECTION_INVALIDE';
    END IF;
  END LOOP;

  IF v_debit = 0 OR v_debit <> v_credit THEN
    RAISE EXCEPTION 'JOURNAL_NON_EQUILIBRE debit=% credit=%', v_debit, v_credit;
  END IF;

  -- Débits un par un, avec garde de solde atomique (ou overdraft autorisé)
  FOR l IN SELECT (line->>'ledger_account_id')::uuid AS ledger_account_id,
                  (line->>'amount_cents')::bigint     AS amount_cents
           FROM jsonb_array_elements(p_lines) AS line
           WHERE (line->>'direction')::text = 'DEBIT'
  LOOP
    UPDATE ledger_account_balance b
       SET balance_cents = b.balance_cents - l.amount_cents, updated_at = now()
      FROM ledger_account a
     WHERE b.ledger_account_id = l.ledger_account_id
       AND a.id = b.ledger_account_id
       AND (a.allow_overdraft OR b.balance_cents >= l.amount_cents);
    IF NOT FOUND THEN
      RAISE EXCEPTION 'FONDS_INSUFFISANTS';
    END IF;
  END LOOP;

  -- Crédits
  FOR l IN SELECT (line->>'ledger_account_id')::uuid AS ledger_account_id,
                  (line->>'amount_cents')::bigint     AS amount_cents
           FROM jsonb_array_elements(p_lines) AS line
           WHERE (line->>'direction')::text = 'CREDIT'
  LOOP
    UPDATE ledger_account_balance
       SET balance_cents = balance_cents + l.amount_cents, updated_at = now()
     WHERE ledger_account_balance.ledger_account_id = l.ledger_account_id;
  END LOOP;

  INSERT INTO journal_entry (id, external_ref, description)
  VALUES (v_entry_id, p_external_ref, p_description);

  INSERT INTO journal_line (journal_entry_id, ledger_account_id, direction, amount_cents)
  SELECT v_entry_id,
         (line->>'ledger_account_id')::uuid,
         (line->>'direction')::text,
         (line->>'amount_cents')::bigint
    FROM jsonb_array_elements(p_lines) AS line;

  RETURN v_entry_id;
END;
